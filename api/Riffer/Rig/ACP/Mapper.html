def elicited_answer(response)
  return nil unless response.is_a?(::ACP::Types::CreateElicitationResponse)

  action = response.action
  return nil unless action.is_a?(::ACP::Types::CreateElicitationResponse::Action::Accept)

  answer = action.content&.fetch(ANSWER_FIELD, nil)
  answer if answer.is_a?(String)
end

def job_exit
  # ACP has no update for a job exit notice, and without a break the replies
  # either side of one read as a single message.
  message_chunk("\n\n")
end

def notice(text)
  # ACP has no update for a notice, and without blank lines around it a
  # notice runs into the reply and into the next notice.
  message_chunk("\n\n#{text}\n\n")
end

def replay_of(message)
  return job_exit if message[Riffer::Rig::Runtime::JobNotification::MARKER]

  case message[:role]
  when 'user' then ::ACP::Types::SessionUpdate::UserMessageChunk.new(content: text_block(message[:content].to_s))
  when 'assistant' then message_chunk(message[:content].to_s) unless message[:content].to_s.strip.empty?
  when 'tool' then replayed_tool_call(message)
  end
end

def stop_reason_of(stop_reason)
  case stop_reason
  when :completed then ::ACP::Types::StopReason::END_TURN
  when Riffer::Rig::Runtime::StopReason::INTERRUPT_CANCELLED then ::ACP::Types::StopReason::CANCELLED
  when :max_steps then ::ACP::Types::StopReason::MAX_TURN_REQUESTS
  else
    stopped = Riffer::Rig::Runtime::StopReason::STOPPED_WITHOUT_COMPLETING.include?(stop_reason)
    stopped ? ::ACP::Types::StopReason::MAX_TOKENS : ::ACP::Types::StopReason::REFUSAL
  end
end

    riffer’s failure reasons (:error, :other, :guardrail_blocked) have no ACP counterpart; refusal is the closest signal that the turn did not complete. @rbs stop_reason: Symbol @rbs return: String

def tool_call(event)

def update_for(event)
  case event
  when Riffer::StreamEvents::TextDelta then message_chunk(event.content)
  when Riffer::Rig::Events::CommandOutput then message_chunk(event.text)
  when Riffer::StreamEvents::ToolCallDone then tool_call(event)
  when Riffer::Rig::Events::AfterToolCall then tool_call_done(event)
  when Riffer::Rig::Events::JobExit then job_exit
  end
end
