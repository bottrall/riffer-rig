def find(host)
  id = resume_id
  if id
    resume(host, id)
  elsif @flags.continue
    loader(host).continue(@flags.to_options)
  end
end

def list(host, all: false)
  loader(host)
    .list(all: all)
    # The id breaks mtime ties so the rows never shuffle under sort_by.
    .sort_by { |header| [header.updated ? -header.updated.to_f : 0.0, header.id] }
    .map { |header| row(header) }
end

def missing(id = nil)

def missing?
  return false unless asked?

  store = @store
  return true unless store

  id = resume_id
  id ? store.read(id).empty? || !gone(id).nil? : store.list(cwd: @cwd).empty?
end

def resume(host, id)
  cwd = saved_cwd(id) || @cwd
  loader(host, cwd).resume(id, @flags.to_options) if File.directory?(cwd)
end
